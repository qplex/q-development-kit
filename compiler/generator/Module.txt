// QDK modules are read-only (attributes belong on engine instances, not the
// module).  This guard overrides the shared module type's setattro, so it must
// identify a QDK module by the private __qdk_module__ marker every one installs
// -- that way it protects every QDK module in the process, not just whichever
// one's PyInit last claimed the shared slot.
static int _q_module_setattro(PyObject *self, PyObject *attr_name, PyObject *value) {
    PyObject *dict = PyModule_GetDict(self);
    if (dict != NULL && PyDict_GetItemString(dict, "__qdk_module__") != NULL) {
        PyErr_SetString(PyExc_AttributeError,
            "module does not support setting attributes; "
            "assign them to an engine instance instead");
        return -1;
    }
    return PyObject_GenericSetAttr(self, attr_name, value);
}

PyMODINIT_FUNC PyInit_@MODULE-NAME(void) {
    PyObject *m;
    @REPEAT(
    if (PyType_Ready(&_@OBJECT-NAME_Type) < 0)
        return NULL;
    @)
    m = PyModule_Create(&moduleDef);
    if (m == NULL)
        return NULL;
    @REPEAT(
    Py_INCREF(&_@OBJECT-NAME_Type);
    if (PyModule_AddObject(m, "@OBJECT-NAME", (PyObject *) &_@OBJECT-NAME_Type) < 0) {
        Py_DECREF(&_@OBJECT-NAME_Type);
        Py_DECREF(m);
        return NULL;
    }
    @)
    PException = PyErr_NewException("@MODULE-NAME.error", NULL, NULL);
    Py_XINCREF(PException);
    if (PyModule_AddObject(m, "error", PException) < 0) {
        Py_XDECREF(PException);
        Py_CLEAR(PException);
        Py_DECREF(m);
        return NULL;
    }

    // Mark this as a QDK module so the read-only guard above recognizes it.
    Py_INCREF(Py_True);
    if (PyModule_AddObject(m, "__qdk_module__", Py_True) < 0) {
        Py_DECREF(Py_True);
        Py_DECREF(m);
        return NULL;
    }
    Py_TYPE(m)->tp_setattro = _q_module_setattro;
    return m;
}
